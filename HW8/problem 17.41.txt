struct FieldInfo{
    int offset;
    int length
}
struct BlockInfo{
    char * blockData;
    BlockInfo * nextBlock;
}
struct FileHeader{
    BlockInfo * startingBlock;
    int recordLength;
    int blockSize;
    FieldInfo * fields;
}

void readField(FileHeader * file, int recordNo, int fieldNo, char * dataOut){
    FieldInfo field = file->fields[fieldNo];
    //get the offset
    int offset = field.offset + recordNo * file->recordLength;
    BlockInfo * startingBlock = file -> startingBlock;
    //find the block we start in
    int fieldStartBlockIndex = offset / file -> blockSize;
    for(int i = 0;i<fieldStartBlockIndex;i++){
        startingBlock = startingBlock->nextBlock;
    }

    //read as much as we can for that block (or all)
    int leftToRead = file->recordLength;
    int inBlockOffset = offset % file -> blockSize;
    int outputOffset = 0;
    while(leftToRead != 0){
        int reading = file->recordLength;
        if(leftToRead < reading){
            reading = leftToRead;
        }
        if(inBlockOffset + reading > file -> blockSize){
            reading = file -> blockSize - inBlockOffset;
        }

        memcpy(dataOut + outputOffset, startingBlock -> blockData + inBlockOffset, reading);
        //read from the next block as necessary until the entire field has been read
        inBlockOffset = 0;
        leftToRead -= reading;
        outputOffset += reading;
        startingBlock = startingBlock->nextBlock;
    }
}

The header contains a pointer to block data that in tern contains a readable pointer to the disk data and a pointer to the next block of the file, if there is one.
There is no separator char as one is not necessary.

struct BlockInfo{
    char * blockData;
    BlockInfo * nextBlock;
}
struct FileHeader{
    BlockInfo * startingBlock;
    int inBlockOffset;
    int blockSize;
    char fieldSeparator;
    char recordSeparator;
}

int readField(FileHeader * file, int recordNo, int fieldNo, char * dataOut){

    BlockInfo * block = file -> startingBlock;
    int offset = file -> inBlockOffset;
    //find the start of the record we want to get
    while(recordNo > 0){
        //if we found the end of this segment
        if(block -> blockData[offset] == file -> recordSeparator){
            recordNo--;//decrease the number we still need to pass
        }

        offset++; //increase the offset
        //if we reached the end of this block
        if(offset >= file -> blockSize){
            //move to the next one
            block = block -> nextBlock;
            offset = 0;
        }
    }

    //find the start of the field we want
    while(fieldNo > 0){
        //if we found the end of this segment
        if(block -> blockData[offset] == file -> fieldSeparator){
            fieldNo--;//decrease the number we still need to pass
        }
        offset++;//increase the offset
        //if we reached the end of this block
        if(offset >= file -> blockSize){
            //move to the next one
            block = block -> nextBlock;
            offset = 0;
        }
    }

    int bytesRead =0;
    while(true){
        char read = block -> blockData[offset];
        if(read == file -> recordSeparator || read == file -> fieldSeparator){
            return bytesRead;
        }
        dataOut[bytesRead] = read;
        offset++;//increase the offset
        bytesRead++;
        //if we reached the end of this block
        if(offset >= file -> blockSize){
            //move to the next one
            block = block -> nextBlock;
            offset = 0;
        }
    }
}

Assumptions:
The passed in file header is valid
The passed in pointer for the output data is provisioned to be at leased the correct size.
The record and field numbers being asked for are not out of range
The header contains a pointer to block data that in tern contains a readable pointer to the disk data, and a pointer to the next block of the file, if there is one.
Each record and each field have specific separator chars that will not naturally appear in the general data
The records might span across multiple blocks

on offset for how far into the given block the file starts

5.6:
a. SELECT name, Major FROM Student JOIN GRADE_REPORT ON Student.Student_number = GRADE_REPORT.Student_number
         GROUP BY Student.Student_number, name, Major
         HAVING count(Grade) = count(CASE WHEN Grade = 'A' THEN 1 END);

b.SELECT name, Major FROM Student JOIN GRADE_REPORT ON Student.Student_number = GRADE_REPORT.Student_number
        GROUP BY Student.Student_number, name, Major
        HAVING count(Grade) != count(CASE WHEN Grade = 'A' THEN 1 END);
